2. 计量项目 - 核心SQL

-- 计量项目查询（关联客户报价） SELECT mi.item_code, mi.item_name, mi.range_min, mi.range_max, mi.accuracy, qp.price FROM bas_meterage_item mi LEFT JOIN bas_quoted_price qp ON mi.item_id = qp.item_id AND qp.customer_id = #{customerId} WHERE mi.del_flag = '0' AND mi.item_name LIKE CONCAT('%',#{itemName},'%') ORDER BY mi.item_code
