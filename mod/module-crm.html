3. 客户管理核心 SQL

-- 客户列表查询 SELECT c.*, cc.cate_name FROM crm_customer c LEFT JOIN crm_customer_cate cc ON c.cate_id = cc.id WHERE c.del_flag = '0' AND c.customer_name LIKE CONCAT('%',#{customerName},'%') AND c.cate_id = #{cateId} ORDER BY c.create_time DESC
